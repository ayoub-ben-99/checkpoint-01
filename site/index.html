package main

import "fmt"

// checknumber returns true if any digit is present.
func checkNumber(s string) bool {
    for _, r := range s {
        if r >= '0' && r <= '9' {
            return true
        }
    }
    return false
}

func main() {
    fmt.Println(checkNumber("hello1")) // true
